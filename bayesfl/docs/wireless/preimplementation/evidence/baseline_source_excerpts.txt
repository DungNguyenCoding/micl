0097 def diagonal_kl(mu_p, var_p, mu_q, var_q) -> np.ndarray:
0098     """Per-coordinate KL(P || Q), evaluated in float64, never a training loss."""
0099     mp, vp, mq, vq = (np.asarray(a, dtype=np.float64) for a in (mu_p, var_p, mu_q, var_q))
0100     if mp.ndim != 1 or mp.size == 0 or any(a.shape != mp.shape for a in (vp, mq, vq)):
0101         raise ValueError("KL requires nonempty aligned vectors")
0102     if any(not np.isfinite(a).all() for a in (mp, vp, mq, vq)) or np.any(vp <= 0) or np.any(vq <= 0):
0103         raise ValueError("KL requires finite means and finite strictly positive variances")
0104     with np.errstate(over="raise", divide="raise", invalid="raise"):
0105         s = np.log(vp) - np.log(vq)
0106         score = 0.5 * (np.expm1(s) - s + (mp - mq) ** 2 / vq)
0107     if not np.isfinite(score).all() or np.any(score < -1e-10):
0108         raise FloatingPointError("Invalid Gaussian KL score")
0109     return np.maximum(score, 0.0)

0022         means = [m[param_idx] for m in means_by_client]
0023         precs = [p[param_idx] for p in precs_by_client]
0024 
0025         if cfg.fola.mode == "paper_reference":
0026             # Released CIFAR implementation:
0027             #   omega_S = sum pi_n omega_n
0028             #   mu_S = sum pi_n omega_n mu_n / (omega_S + eps)
0029             denom = np.zeros_like(precs[0], dtype=np.float64)
0030             numer = np.zeros_like(means[0], dtype=np.float64)
0031             for mu_n, omega_n, weight in zip(means, precs, weights):
0032                 omega64 = np.asarray(omega_n, dtype=np.float64)
0033                 w = float(weight)
0034                 denom += w * omega64
0035                 numer += w * omega64 * np.asarray(mu_n, dtype=np.float64)
0036             mu = numer / (denom + float(cfg.fola.aggregation_epsilon))
0037             precision = denom
0038             global_means.append(mu.astype(means[0].dtype, copy=False))
0039             global_precs.append(precision.astype(precs[0].dtype, copy=False))
0040         else:
0041             mu, precision = gaussian_product(
0042                 means,
0043                 precs,
0044                 weights,
0045                 precision_min=cfg.fola.precision_min,
0046                 precision_max=cfg.fola.precision_max,
0047             )
0048             global_means.append(mu)
0049             global_precs.append(precision)
